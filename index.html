window.callMilloraAI = async function(prompt) { if (typeof prompt !== "string" || !prompt.trim()) { throw new Error("Enter a description for Millora AI first."); } const { data: sessionData, error: sessionError } = await supabase.auth.getSession(); if (sessionError) { throw new Error("Couldn't check your login: " + sessionError.message); } const session = sessionData?.session; if (!session) { throw new Error("Please log in to use Millora AI."); } const { data, error } = await supabase.functions.invoke( "bright-worker", { body: { prompt: prompt.trim() }, headers: { Authorization: "Bearer " + session.access_token } } ); if (error) { let detail = error.message || "The AI request failed."; try { const response = error.context; if (response && typeof response.json === "function") { const body = await response.clone().json(); detail = body?.error || body?.message || body?.details || detail; } else if (response && typeof response.text === "function") { const bodyText = await response.clone().text(); if (bodyText) detail = bodyText; } } catch (_) {} throw new Error(detail); } let generatedHTML = typeof data === "string" ? data : data?.html ?? data?.content ?? data?.output ?? ""; if (typeof generatedHTML !== "string" || !generatedHTML.trim()) { throw new Error( data?.error || "Millora AI returned no HTML. Check bright-worker's response." ); } generatedHTML = generatedHTML.trim() .replace(/^```(?:html)?\s*/i, "") .replace(/\s*```$/, ""); if (!/<(?:!doctype\s+html|html)(?:\s|>)/i.test(generatedHTML)) { throw new Error( "Millora AI responded, but didn't return a complete HTML page." ); } return generatedHTML; };
